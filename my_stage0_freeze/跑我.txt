@echo off
chcp 936 >nul
cd /d "%~dp0"
title 看懂求解器

echo ============================================================
echo   第 1 步 / 共 3 步  检查 Python 有没有装
echo ============================================================
python --version
if errorlevel 1 goto NOPYTHON
echo   [OK] Python 找到了
echo.

echo ============================================================
echo   第 2 步 / 共 3 步  检查 pyscipopt 有没有装
echo ============================================================
python -c "import pyscipopt; print('   [OK] pyscipopt', pyscipopt.__version__)" 2>nul
if errorlevel 1 goto INSTALL
goto RUN

:INSTALL
echo   [没装] 正在自动安装，第一次会下载几十兆，请等 1-3 分钟...
echo.
python -m pip install pyscipopt
if errorlevel 1 goto PIPFAIL
echo.
echo   [OK] 装好了
goto RUN

:RUN
echo.
echo ============================================================
echo   第 3 步 / 共 3 步  开始运行
echo ============================================================
echo.
python "看懂求解器.py"
echo.
echo ============================================================
echo   跑完了。往上滚动可以看全部输出。
echo   按任意键关闭这个窗口。
echo ============================================================
pause >nul
exit /b

:NOPYTHON
echo.
echo   [失败] 没有找到 Python。
echo.
echo   请先去 https://www.python.org/downloads/ 下载安装，
echo   安装时务必勾选最下面那个 "Add python.exe to PATH"。
echo   装完后重启电脑，再双击本文件。
echo.
pause
exit /b

:PIPFAIL
echo.
echo   [失败] pyscipopt 安装失败。
echo   常见原因: 没联网 / 公司网络挡了 / Python 版本太老。
echo   可以试试在命令行手动运行:  python -m pip install pyscipopt
echo.
pause
exit /b
